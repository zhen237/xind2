```python
"""通信基站智能设计平台 — QGIS Plugin 入口"""
from qgis.gui import QgisInterface
from PyQt5.QtWidgets import QAction, QToolBar
from PyQt5.QtGui import QIcon
import os

class BaseStationDesignPlugin:
    """插件主类"""
    
    def __init__(self, iface: QgisInterface):
        self.iface = iface
        self.plugin_dir = os.path.dirname(__file__)
        self.actions = []
        self.toolbar = None
        self.design_dock = None
        self.bom_dock = None
    
    def initGui(self):
        """初始化插件UI — 在QGIS菜单和工具栏注册"""
        icon_path = os.path.join(self.plugin_dir, 'resources', 'icon.png')
        
        # 智能设计面板
        design_action = QAction(
            QIcon(icon_path),
            '基站智能设计',
            self.iface.mainWindow()
        )
        design_action.triggered.connect(self.open_design_panel)
        self.iface.addPluginToMenu('&通信基建设计', design_action)
        self.iface.addToolBarIcon(design_action)
        self.actions.append(design_action)
        
        # BOM生成面板
        bom_action = QAction(
            QIcon(icon_path),
            'BOM生成',
            self.iface.mainWindow()
        )
        bom_action.triggered.connect(self.open_bom_panel)
        self.iface.addPluginToMenu('&通信基建设计', bom_action)
        self.iface.addToolBarIcon(bom_action)
        self.actions.append(bom_action)
    
    def unload(self):
        """卸载插件 — 清理所有UI元素"""
        for action in self.actions:
            self.iface.removePluginMenu('&通信基建设计', action)
            self.iface.removeToolBarIcon(action)
        if self.design_dock:
            self.iface.removeDockWidget(self.design_dock)
        if self.bom_dock:
            self.iface.removeDockWidget(self.bom_dock)
    
    def open_design_panel(self):
        """打开基站智能设计面板（子赛题1）"""
        from dock_widget import DesignDockWidget
        if self.design_dock is None:
            self.design_dock = DesignDockWidget(self.iface)
            self.iface.addDockWidget(4, self.design_dock)  # 4=RightDockWidgetArea
        self.design_dock.show()
    
    def open_bom_panel(self):
        """打开BOM生成面板（子赛题4）"""
        from bom_generator.bom_dialog import BomDialog
        dialog = BomDialog(self.iface)
        dialog.exec_()


def classFactory(iface):
    """QGIS插件入口点"""
    return BaseStationDesignPlugin(iface)
```

"""站点数据模型"""
from dataclasses import dataclass, field
from typing import List, Optional

@dataclass
class Site:
    """通信基站站点"""
    site_id: str                           # 站点编码，如 "BTS-WH-001"
    name: str                             # 站点名称
    longitude: float                      # 经度 (EPSG:4326)
    latitude: float                       # 纬度 (EPSG:4326)
    site_type: str = "MACRO"              # MACRO/SMALL/INDOOR
    tower_type: str = "MONOPOLE"          # MONOPOLE/LATTICE/TOWER
    tower_height: float = 35.0            # 塔高 (m)
    scenario: str = "URBAN"               # URBAN/SUBURBAN/RURAL
    status: str = "PLANNED"               # PLANNED/APPROVED/BUILT
    antennas: List['Antenna'] = field(default_factory=list)
    properties: dict = field(default_factory=dict)  # 扩展属性
    
    def to_geojson_feature(self) -> dict:
        """转为GeoJSON Feature"""
        from .antenna import Antenna
        return {
            "type": "Feature",
            "geometry": {
                "type": "Point",
                "coordinates": [self.longitude, self.latitude]
            },
            "properties": {
                "siteId": self.site_id,
                "name": self.name,
                "siteType": self.site_type,
                "towerType": self.tower_type,
                "towerHeight": self.tower_height,
                "scenario": self.scenario,
                "antennas": [a.to_dict() for a in self.antennas],
                **self.properties
            }
        }
    
    @classmethod
    def from_geojson_feature(cls, feature: dict) -> 'Site':
        """从GeoJSON Feature解析"""
        from .antenna import Antenna
        props = feature.get("properties", {})
        coords = feature.get("geometry", {}).get("coordinates", [0, 0])
        return cls(
            site_id=props.get("siteId", ""),
            name=props.get("name", ""),
            longitude=coords[0],
            latitude=coords[1],
            site_type=props.get("siteType", "MACRO"),
            tower_type=props.get("towerType", "MONOPOLE"),
            tower_height=props.get("towerHeight", 35.0),
            scenario=props.get("scenario", "URBAN"),
            antennas=[Antenna.from_dict(a) for a in props.get("antennas", [])],
            properties={k: v for k, v in props.items() 
                       if k not in ("siteId","name","siteType","towerType","towerHeight","scenario","antennas")}
        )

"""蜂窝拓扑站址布局生成"""
import math
from typing import List, Tuple, Optional
from shapely.geometry import Point, Polygon, box
from models.site import Site
from models.antenna import Antenna
from .rules import BAND_CONFIGS, DEFAULT_SITE_PARAMS, BandConfig

def generate_hex_grid(
    bbox: Tuple[float, float, float, float],  # (min_lon, min_lat, max_lon, max_lat)
    isr_km: float,
    rotation_deg: float = 0.0,
) -> List[Tuple[float, float]]:
    """
    生成六边形网格中心点列表。
    采用偏移行法(offset rows)生成整齐六边形。
    """
    lon_min, lat_min, lon_max, lat_max = bbox
    # 将km转为经纬度近似值（中纬度地区1°≈111km lon, 111*cos(30.5°)≈95.7km lat）
    lon_per_km = 1.0 / (111.0 * math.cos(math.radians((lat_min + lat_max) / 2)))
    lat_per_km = 1.0 / 111.0
    
    isr_lon = isr_km * lon_per_km
    isr_lat = isr_km * lat_per_km
    
    # 六边形列间距 = sqrt(3)*ISR, 行交替偏移 ISR/2
    col_spacing = math.sqrt(3) * isr_lon
    row_spacing = 1.5 * isr_lat
    
    centers = []
    row = 0
    y = lat_min
    while y <= lat_max + row_spacing:
        x_offset = (isr_lon / 2) if row % 2 == 1 else 0
        x = lon_min + x_offset
        while x <= lon_max + col_spacing:
            centers.append((x, y))
            x += col_spacing
        y += row_spacing
        row += 1
    
    return centers


def generate_sites_from_grid(
    grid_centers: List[Tuple[float, float]],
    band_config: BandConfig,
    site_type: str = "MACRO",
    tower_height: float = 35.0,
    scenario: str = "URBAN",
    num_sectors: int = 3,
    existing_sites: Optional[List[Site]] = None,
) -> List[Site]:
    """
    从网格中心点生成Site列表，自动配置天线参数。
    排除与已有站点重叠的中心点。
    """
    existing_coords = set()
    if existing_sites:
        for es in existing_sites:
            existing_coords.add((round(es.longitude, 6), round(es.latitude, 6)))
    
    sites = []
    for i, (lon, lat) in enumerate(grid_centers):
        # 跳过与已有站点过近的位置（< 100m）
        too_close = False
        for ex_lon, ex_lat in existing_coords:
            dist = math.sqrt((lon - ex_lon)**2 + (lat - ex_lat)**2) * 111000
            if dist < 100:
                too_close = True
                break
        if too_close:
            continue
        
        site = Site(
            site_id=f"BTS-{scenario[:4].upper()}-{i+1:03d}",
            name=f"{scenario}-{i+1:03d}",
            longitude=lon,
            latitude=lat,
            site_type=site_type,
            tower_height=tower_height,
            scenario=scenario,
        )
        
        # 配置定向天线（按扇区数均匀分布）
        for s in range(num_sectors):
            azimuth = (360.0 / num_sectors) * s
            antenna = Antenna(
                antenna_type="AAU5313" if band_config.frequency_mhz > 2000 else "RRU5301",
                azimuth=azimuth,
                height=tower_height,
                band=f"{band_config.frequency_mhz}MHz",
            )
            site.antennas.append(antenna)
        
        sites.append(site)
    
    return sites

"""覆盖范围计算 — Okumura-Hata 传播模型"""
import math
from typing import List, Tuple
from shapely.geometry import Point
import numpy as np

def okumura_hata_path_loss(
    frequency_mhz: float,
    distance_km: float,
    tx_height_m: float,
    rx_height_m: float = 1.5,
    environment: str = "URBAN",
) -> float:
    """
    计算Okumura-Hata模型路径损耗 (dB)。
    
    Args:
        frequency_mhz: 载波频率 (150-2000 MHz，外推至3500)
        distance_km: 收发距离 (1-20 km)
        tx_height_m: 发射天线有效高度 (30-200 m)
        rx_height_m: 接收天线高度 (1-10 m)
        environment: URBAN/SUBURBAN/RURAL
    """
    if environment == "URBAN":
        # 大城市修正
        if frequency_mhz <= 200:
            a_hr = 8.29 * (math.log10(1.54 * rx_height_m))**2 - 1.1
        else:
            a_hr = 3.2 * (math.log10(11.75 * rx_height_m))**2 - 4.97
    else:
        a_hr = (1.1 * math.log10(frequency_mhz) - 0.7) * rx_height_m \
               - (1.56 * math.log10(frequency_mhz) - 0.8)
    
    L_urban = (69.55 + 26.16 * math.log10(frequency_mhz)
               - 13.82 * math.log10(tx_height_m)
               + (44.9 - 6.55 * math.log10(tx_height_m)) * math.log10(distance_km)
               - a_hr)
    
    if environment == "SUBURBAN":
        L = L_urban - 2 * (math.log10(frequency_mhz / 28))**2 - 5.4
    elif environment == "RURAL":
        L = L_urban - 4.78 * (math.log10(frequency_mhz))**2 + 18.33 * math.log10(frequency_mhz) - 40.94
    else:
        L = L_urban
    
    return L


def calculate_rsrp(
    tx_power_dbm: float,   # 发射功率 dBm
    tx_gain_dbi: float,    # 发射天线增益 dBi
    path_loss_db: float,   # 路径损耗 dB
    rx_gain_dbi: float = 0.0,
    shadow_fade_db: float = 8.0,
) -> float:
    """计算 RSRP (dBm)"""
    return tx_power_dbm + tx_gain_dbi - path_loss_db + rx_gain_dbi - shadow_fade_db


def generate_coverage_raster(
    site_lon: float,
    site_lat: float,
    tx_height_m: float,
    frequency_mhz: float,
    tx_power_w: float,
    antenna_gain_dbi: float,
    radius_km: float = 2.0,
    resolution_m: int = 50,
    rsrp_threshold_dbm: float = -110,
) -> dict:
    """
    生成单个站点的覆盖栅格数据。
    返回GeoJSON格式的热力图数据。
    """
    tx_power_dbm = 10 * math.log10(tx_power_w * 1000)
    
    lon_per_km = 1.0 / (111.0 * math.cos(math.radians(site_lat)))
    lat_per_km = 1.0 / 111.0
    
    points = []
    steps = int(radius_km * 1000 / resolution_m)
    
    for i in range(-steps, steps + 1):
        for j in range(-steps, steps + 1):
            d_lon = i * resolution_m / 1000 * lon_per_km
            d_lat = j * resolution_m / 1000 * lat_per_km
            d_km = math.sqrt((i * resolution_m / 1000)**2 + (j * resolution_m / 1000)**2)
            
            if d_km > radius_km:
                continue
            
            path_loss = okumura_hata_path_loss(frequency_mhz, max(d_km, 0.01), tx_height_m)
            rsrp = calculate_rsrp(tx_power_dbm, antenna_gain_dbi, path_loss)
            
            if rsrp >= rsrp_threshold_dbm:
                points.append({
                    "type": "Feature",
                    "geometry": {
                        "type": "Point",
                        "coordinates": [site_lon + d_lon, site_lat + d_lat]
                    },
                    "properties": {
                        "rsrp": round(rsrp, 1),
                        "distance_km": round(d_km, 3),
                    }
                })
    
    return {
        "type": "FeatureCollection",
        "features": points,
        "metadata": {
            "siteLocation": [site_lon, site_lat],
            "frequencyMHz": frequency_mhz,
            "rsrpThresholdDbm": rsrp_threshold_dbm,
            "resolutionM": resolution_m,
        }
    }

"""障碍物避让 — 基于OSM/土地利用数据的站点选址约束"""
from typing import List, Tuple
from shapely.geometry import Point, Polygon, shape
import json

class AvoidanceChecker:
    """站点选址避让检查器"""
    
    def __init__(self):
        self.avoidance_polygons = []  # List[(geometry, rule_name, buffer_m)]
    
    def load_osm_buildings(self, geojson_path: str, buffer_m: float = 20.0):
        """加载OSM建筑轮廓作为避让区域"""
        with open(geojson_path, 'r', encoding='utf-8') as f:
            geojson = json.load(f)
        for feature in geojson.get('features', []):
            geom = shape(feature['geometry'])
            buffered = geom.buffer(buffer_m / 111000.0)  # 近似转度
            self.avoidance_polygons.append((buffered, "建筑避让", buffer_m))
    
    def load_water_bodies(self, geojson_path: str, buffer_m: float = 50.0):
        """加载水体区域"""
        with open(geojson_path, 'r', encoding='utf-8') as f:
            geojson = json.load(f)
        for feature in geojson.get('features', []):
            geom = shape(feature['geometry'])
            buffered = geom.buffer(buffer_m / 111000.0)
            self.avoidance_polygons.append((buffered, "水体避让", buffer_m))
    
    def is_site_valid(self, lon: float, lat: float) -> Tuple[bool, List[str]]:
        """
        检查站点位置是否有效。
        返回 (是否有效, 冲突原因列表)
        """
        point = Point(lon, lat)
        conflicts = []
        for geom, rule_name, buffer_m in self.avoidance_polygons:
            if geom.contains(point):
                conflicts.append(f"{rule_name}（{buffer_m}m缓冲区）")
        return (len(conflicts) == 0, conflicts)
    
    def filter_valid_sites(self, sites: List[Tuple[float, float]]) -> List[Tuple[float, float]]:
        """过滤出有效的站点坐标"""
        valid = []
        for lon, lat in sites:
            ok, reasons = self.is_site_valid(lon, lat)
            if ok:
                valid.append((lon, lat))
        return valid

from typing import List
from models.site import Site
from bom_generator.bom_model import BomItem

class BaseExtractor:
    """BOM物料提取器基类"""
    category: str = "OTHER"
    
    def extract(self, site: Site) -> List[BomItem]:
        """从单个站点提取物料清单"""
        raise NotImplementedError
    
    def extract_batch(self, sites: List[Site]) -> List[BomItem]:
        """批量提取，自动合并相同物料"""
        all_items = []
        for site in sites:
            all_items.extend(self.extract(site))
        return self._merge_same_items(all_items)
    
    def _merge_same_items(self, items: List[BomItem]) -> List[BomItem]:
        """合并相同物料编码的数量"""
        merged = {}
        for item in items:
            if item.item_code in merged:
                merged[item.item_code].quantity += item.quantity
                merged[item.item_code].total_price += item.total_price
            else:
                merged[item.item_code] = item
        return list(merged.values())

package com.comm.m04.review.service.impl;

import com.comm.m04.review.service.ReviewEngineService;
import org.springframework.stereotype.Component;
import java.util.*;

@Component
public class ElectricityCrossingReviewer implements ReviewEngineService {

    @Override
    public String getDimension() {
        return "ELECTRICITY";
    }

    @Override
    public List<Map<String, Object>> review(Map<String, Object> designData, Long projectId) {
        List<Map<String, Object>> results = new ArrayList<>();
        
        // 从设计数据中提取站点列表
        List<Map<String, Object>> features = extractFeatures(designData);
        
        for (Map<String, Object> feature : features) {
            Map<String, Object> props = getProperties(feature);
            List<Double> coords = getCoordinates(feature);
            if (coords == null || coords.size() < 2) continue;
            
            double siteLon = coords.get(0);
            double siteLat = coords.get(1);
            double towerHeight = getDouble(props, "towerHeight", 35.0);
            
            // 规则ELEC-001: 电力线水平安全距离 ≥ 1.5×塔高
            double minDistance = 1.5 * towerHeight;
            // 此处需要实际电力线数据。Phase 0阶段从OSM电力线图层获取
            // 如果无电力线数据，返回空（无冲突）
            
            // Mock: 假设光谷大道有一条10kV电力线穿过
            // 114.395, 30.510 附近
            double powerLineLon = 114.395;
            double powerLineLat = 30.510;
            double actualDistance = haversineDistance(siteLon, siteLat, powerLineLon, powerLineLat);
            
            if (actualDistance < minDistance) {
                results.add(buildResult(
                    "ELEC-001",
                    "电力线水平安全距离违规",
                    "WARNING",
                    1,
                    siteLon, siteLat,
                    String.format("站点距电力线 %.1fm，要求 ≥ %.1fm（1.5×%.1fm塔高）", 
                        actualDistance, minDistance, towerHeight),
                    "建议将站点水平移动至少" + (minDistance - actualDistance) + "m",
                    String.format("%.1fm", actualDistance),
                    String.format("≥ %.1fm", minDistance)
                ));
            }
        }
        
        return results;
    }
    
    @SuppressWarnings("unchecked")
    private List<Map<String, Object>> extractFeatures(Map<String, Object> data) {
        Object features = data.get("features");
        if (features instanceof List) {
            return (List<Map<String, Object>>) features;
        }
        return Collections.emptyList();
    }
    
    @SuppressWarnings("unchecked")
    private Map<String, Object> getProperties(Map<String, Object> feature) {
        Object props = feature.get("properties");
        return (props instanceof Map) ? (Map<String, Object>) props : Collections.emptyMap();
    }
    
    @SuppressWarnings("unchecked")
    private List<Double> getCoordinates(Map<String, Object> feature) {
        Object geom = feature.get("geometry");
        if (!(geom instanceof Map)) return null;
        Object coords = ((Map<String, Object>) geom).get("coordinates");
        return (coords instanceof List) ? (List<Double>) coords : null;
    }
    
    private double getDouble(Map<String, Object> map, String key, double defaultValue) {
        Object val = map.get(key);
        if (val instanceof Number) return ((Number) val).doubleValue();
        return defaultValue;
    }
    
    private double haversineDistance(double lon1, double lat1, double lon2, double lat2) {
        double R = 6371000; // 地球半径 (m)
        double dLat = Math.toRadians(lat2 - lat1);
        double dLon = Math.toRadians(lon2 - lon1);
        double a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
                   Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2)) *
                   Math.sin(dLon / 2) * Math.sin(dLon / 2);
        double c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
        return R * c;
    }
    
    private Map<String, Object> buildResult(
            String ruleCode, String ruleName, String status, 
            int severity, double lon, double lat,
            String description, String suggestion,
            String measuredValue, String thresholdValue) {
        Map<String, Object> result = new LinkedHashMap<>();
        result.put("dimension", getDimension());
        result.put("ruleCode", ruleCode);
        result.put("ruleName", ruleName);
        result.put("status", status);
        result.put("severity", severity);
        result.put("locationGeojson", String.format(
            "{\"type\":\"Point\",\"coordinates\":[%f,%f]}", lon, lat));
        result.put("description", description);
        result.put("suggestion", suggestion);
        result.put("measuredValue", measuredValue);
        result.put("thresholdValue", thresholdValue);
        return result;
    }
}

```python
"""安全帽检测器 — 基于YOLOv8"""
from pathlib import Path
from typing import List, Optional
from ultralytics import YOLO
import numpy as np
from PIL import Image

class HelmetDetector:
    """安全帽佩戴检测"""
    
    def __init__(self, model_path: Optional[str] = None):
        if model_path and Path(model_path).exists():
            self.model = YOLO(model_path)
            self.loaded = True
        else:
            self.model = None
            self.loaded = False
            print("[WARN] HelmetDetector: model file not found, using mock")
    
    def detect(self, image: np.ndarray, confidence_threshold: float = 0.5) -> List[dict]:
        """
        检测图片中的安全帽违规。
        返回违规列表。
        """
        if not self.loaded:
            # Mock fallback — 返回空（无违规）
            return []
        
        results = self.model(image, conf=confidence_threshold)
        violations = []
        
        for result in results:
            boxes = result.boxes
            if boxes is None:
                continue
            for box in boxes:
                cls_id = int(box.cls[0])
                cls_name = self.model.names[cls_id] if hasattr(self.model, 'names') else f"class_{cls_id}"
                conf = float(box.conf[0])
                xyxy = box.xyxy[0].tolist()
                
                violations.append({
                    "type": self._map_class_to_violation(cls_name),
                    "severity": 1,
                    "bbox": {
                        "x": int(xyxy[0]),
                        "y": int(xyxy[1]),
                        "w": int(xyxy[2] - xyxy[0]),
                        "h": int(xyxy[3] - xyxy[1]),
                    },
                    "confidence": round(conf, 4),
                    "suggestion": self._get_suggestion(cls_name),
                })
        
        return violations
    
    def _map_class_to_violation(self, class_name: str) -> str:
        mapping = {
            "no_helmet": "NO_HELMET",
            "no_vest": "NO_VEST",
            "person_no_helmet": "NO_HELMET",
        }
        return mapping.get(class_name, class_name.upper())
    
    def _get_suggestion(self, class_name: str) -> str:
        suggestions = {
            "no_helmet": "检测到未佩戴安全帽人员，请立即纠正",
            "no_vest": "检测到未穿反光背心人员，请穿戴后进入",
        }
        return suggestions.get(class_name, f"检测到违规行为: {class_name}")

package com.comm.m04.review.service.impl;

import com.comm.m04.review.service.ReviewEngineService;
import org.springframework.stereotype.Component;
import java.util.*;

@Component
public class EmfComplianceReviewer implements ReviewEngineService {

    private static final double PUBLIC_EXPOSURE_LIMIT_W_M2 = 0.4;  // GB 8702 公众暴露限值

    @Override
    public String getDimension() { return "EMF"; }

    @Override
    @SuppressWarnings("unchecked")
    public List<Map<String, Object>> review(Map<String, Object> designData, Long projectId) {
        List<Map<String, Object>> results = new ArrayList<>();
        List<Map<String, Object>> features = getList(designData, "features");

        for (Map<String, Object> feature : features) {
            Map<String, Object> props = getMap(feature, "properties");
            List<Map<String, Object>> antennas = getList(props, "antennas");
            List<Double> coords = getCoordList(feature);

            for (Map<String, Object> ant : antennas) {
                double power = getDouble(ant, "power", 200.0);
                double gain = getDouble(ant, "gain", 24.0);
                double eirp = power * Math.pow(10, gain / 10.0);  // EIRP in W

                // 计算安全距离（远场近似：功率密度 = EIRP / (4πR²)）
                double safeDistance = Math.sqrt(eirp / (4 * Math.PI * PUBLIC_EXPOSURE_LIMIT_W_M2));

                if (safeDistance > 20.0) {  // 安全距离 > 20m 需标记
                    results.add(Map.of(
                        "dimension", "EMF",
                        "ruleCode", "EMF-001",
                        "ruleName", "公众暴露电磁辐射限值",
                        "status", "WARNING",
                        "severity", 2,
                        "locationGeojson", buildPointGeoJson(coords),
                        "description", String.format(
                            "天线 EIRP=%.1fW，公众安全距离 %.1fm > 20m，需评估周边敏感区域",
                            eirp, safeDistance),
                        "suggestion", "建议核查半径" + Math.round(safeDistance) + "m内是否有学校/医院/居民区",
                        "measuredValue", String.format("%.1fW EIRP", eirp),
                        "thresholdValue", String.format("安全距离 ≤ 20m")
                    ));
                }
            }
        }
        return results;
    }

    @SuppressWarnings("unchecked")
    private List<Map<String, Object>> getList(Map<String, Object> map, String key) {
        Object v = map.get(key);
        return (v instanceof List) ? (List<Map<String, Object>>) v : Collections.emptyList();
    }
    @SuppressWarnings("unchecked")
    private Map<String, Object> getMap(Map<String, Object> map, String key) {
        Object v = map.get(key);
        return (v instanceof Map) ? (Map<String, Object>) v : Collections.emptyMap();
    }

        if (!(geom instanceof Map)) return Collections.emptyList();
        Object coords = ((Map<String, Object>) geom).get("coordinates");
        return (coords instanceof List) ? (List<Double>) coords : Collections.emptyList();
    }

package com.comm.m04.review.service.impl;

import com.comm.m04.review.service.ReviewEngineService;
import org.springframework.stereotype.Component;
import java.util.*;

@Component
public class SpatialConflictReviewer implements ReviewEngineService {

    private static final double SAME_BAND_MIN_DISTANCE_M = 200.0;
    private static final double DIFF_BAND_MIN_DISTANCE_M = 100.0;

    @Override
    public String getDimension() { return "SPATIAL"; }

    @Override
    @SuppressWarnings("unchecked")
    public List<Map<String, Object>> review(Map<String, Object> designData, Long projectId) {
        List<Map<String, Object>> results = new ArrayList<>();
        List<Map<String, Object>> features = new ArrayList<>();
        Object f = designData.get("features");
        if (f instanceof List) features = (List<Map<String, Object>>) f;

        // 提取所有站点的位置和频段
        List<SiteInfo> sites = new ArrayList<>();
        for (Map<String, Object> feat : features) {
            Map<String, Object> props = feat.get("properties") instanceof Map ? 
                (Map<String, Object>) feat.get("properties") : Collections.emptyMap();
            List<Double> coords = extractCoords(feat);
            if (coords.size() < 2) continue;
            String firstBand = "";
            List<Map<String, Object>> ants = props.get("antennas") instanceof List ?
                (List<Map<String, Object>>) props.get("antennas") : Collections.emptyList();
            if (!ants.isEmpty()) firstBand = String.valueOf(ants.get(0).getOrDefault("band", ""));
            sites.add(new SiteInfo(
                String.valueOf(props.getOrDefault("siteId", "")),
                coords.get(0), coords.get(1),
                firstBand
            ));
        }

        // 两两比较
        for (int i = 0; i < sites.size(); i++) {
            for (int j = i + 1; j < sites.size(); j++) {
                SiteInfo a = sites.get(i), b = sites.get(j);
                double dist = haversineM(a.lon, a.lat, b.lon, b.lat);
                boolean sameBand = a.band.equals(b.band);
                double minDist = sameBand ? SAME_BAND_MIN_DISTANCE_M : DIFF_BAND_MIN_DISTANCE_M;

                if (dist < minDist) {
                    results.add(Map.of(
                        "dimension", "SPATIAL",
                        "ruleCode", sameBand ? "SPAT-001" : "SPAT-002",
                        "ruleName", sameBand ? "同频站点间距" : "异频站点间距",
                        "status", "FAIL",
                        "severity", 1,
                        "locationGeojson", String.format(
                            "{\"type\":\"Point\",\"coordinates\":[%f,%f]}", a.lon, a.lat),
                        "description", String.format(
                            "%s ↔ %s 距离 %.0fm < %s最低间距 %.0fm",
                            a.siteId, b.siteId, dist, sameBand ? "同频" : "异频", minDist),
                        "suggestion", "建议调整站点位置或合并为共址站",
                        "measuredValue", String.format("%.0fm", dist),
                        "thresholdValue", String.format("≥ %.0fm", minDist)
                    ));
                }
            }
        }
        return results;
    }

    @SuppressWarnings("unchecked")
    private List<Double> extractCoords(Map<String, Object> feature) {
        Object geom = feature.get("geometry");
        if (geom instanceof Map) {
            Object coords = ((Map<String, Object>) geom).get("coordinates");
            if (coords instanceof List) return (List<Double>) coords;
        }
        return Collections.emptyList();
    }
    private double haversineM(double lon1, double lat1, double lon2, double lat2) {
        double R = 6371000, dLat = Math.toRadians(lat2 - lat1), dLon = Math.toRadians(lon2 - lon1);
        double a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
                   Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2)) *
                   Math.sin(dLon / 2) * Math.sin(dLon / 2);
        return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    }
    private static class SiteInfo {
        String siteId, band; double lon, lat;
        SiteInfo(String i, double lo, double la, String b) { siteId=i; lon=lo; lat=la; band=b; }
    }
}

package com.comm.m04.review.service.impl;

import com.comm.m04.review.entity.DesignReviewTask;
import com.comm.m04.review.entity.ReviewResult;
import com.comm.m04.review.mapper.DesignReviewTaskMapper;
import com.comm.m04.review.mapper.ReviewResultMapper;
import com.comm.m04.review.service.ReviewEngineService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import java.time.LocalDateTime;
import java.util.*;

@Service
@RequiredArgsConstructor
public class ReviewEngineServiceImpl {

    private final List<ReviewEngineService> reviewers;  // Spring自动注入所有ReviewEngineService实现
    private final DesignReviewTaskMapper taskMapper;
    private final ReviewResultMapper resultMapper;

    /**
     * 执行全维度审查（4个Reviewer并行编排）。
     * @param designData GeoJSON设计数据（已解析为Map）
     * @param projectId 项目ID
     * @param taskId 审查任务ID
     * @return 所有审查结果
     */
    public List<Map<String, Object>> executeFullReview(
            Map<String, Object> designData, Long projectId, Long taskId) {
        
        // 更新任务状态为"审查中"
        DesignReviewTask task = new DesignReviewTask();
        task.setId(taskId);
        task.setStatus(1);  // 审查中
        taskMapper.updateById(task);

        List<Map<String, Object>> allResults = new ArrayList<>();

        for (ReviewEngineService reviewer : reviewers) {
            List<Map<String, Object>> dimResults = reviewer.review(designData, projectId);
            allResults.addAll(dimResults);

            // 写入数据库
            for (Map<String, Object> r : dimResults) {
                ReviewResult entity = new ReviewResult();
                entity.setTaskId(taskId);
                entity.setDimension(String.valueOf(r.getOrDefault("dimension", "")));
                entity.setRuleCode(String.valueOf(r.getOrDefault("ruleCode", "")));
                entity.setRuleName(String.valueOf(r.getOrDefault("ruleName", "")));
                entity.setStatus("PASS".equals(r.get("status")) ? 0 : 
                                 "WARNING".equals(r.get("status")) ? 1 : 2);
                entity.setSeverity((Integer) r.getOrDefault("severity", 3));
                entity.setLocationGeojson(String.valueOf(r.getOrDefault("locationGeojson", "")));
                entity.setDescription(String.valueOf(r.getOrDefault("description", "")));
                entity.setSuggestion(String.valueOf(r.getOrDefault("suggestion", "")));
                entity.setMeasuredValue(String.valueOf(r.getOrDefault("measuredValue", "")));
                entity.setThresholdValue(String.valueOf(r.getOrDefault("thresholdValue", "")));
                entity.setCreateTime(LocalDateTime.now());
                resultMapper.insert(entity);
            }
        }

        // 更新任务状态
        boolean hasFailure = allResults.stream().anyMatch(r -> "FAIL".equals(r.get("status")));
        task.setStatus(hasFailure ? 3 : 2);  // 3=有风险, 2=已完成
        taskMapper.updateById(task);

        return allResults;
    }
}

import json

def _load_config() -> dict:
    """加载QGIS插件配置"""
    if CONFIG_PATH.exists():
        with open(CONFIG_PATH, "r", encoding="utf-8") as f:

def push_bom_to_m04(bom: BomDocument) -> dict:
    """
    将BOM文档推送到M04后端 (API-04)。
    返回: {"bomId": 789, "totalItems": 42, ...}
    """
    config = _load_config()
    url = f"{config['m04_base_url']}/api/m04/bom/upload"
    
    payload = json.dumps(bom.to_api_payload()).encode("utf-8")
    
    req = urllib.request.Request(url, data=payload, method="POST")
    req.add_header("Content-Type", "application/json")
    
    token = config.get("jwt_token", "")
    if token:
        req.add_header("Authorization", f"Bearer {token}")
    
    try:
        with urllib.request.urlopen(req, timeout=config["timeout_seconds"]) as resp:
            result = json.loads(resp.read().decode("utf-8"))
            if result.get("code") == 200:
                return result.get("data", {})
            raise Exception(f"API返回错误: {result.get('message', '未知错误')}")
    except urllib.error.HTTPError as e:
        raise Exception(f"HTTP {e.code}: {e.reason} — 请确认M04后端已启动 ({config['m04_base_url']})")
    except urllib.error.URLError as e:
        raise Exception(f"连接失败: {e.reason} — 请确认M04后端地址: {config['m04_base_url']}")

"""视距分析 — 基于DEM的可见性计算"""
import math
from typing import List, Tuple

def calculate_viewshed_radius(
    tower_height_m: float,
    dem_resolution_m: float = 30.0,
    earth_curvature_correction: bool = True,
) -> float:
    """
    计算理论视距半径 (km)。
    考虑地球曲率：d = sqrt(2Rh + h²) ≈ sqrt(2Rh) = 3.57 * sqrt(h)
    """
    # 大气折射修正系数
    K = 4.0 / 3.0 if earth_curvature_correction else 1.0
    R_earth_km = 6371.0 * K
    distance_km = math.sqrt(2 * R_earth_km * tower_height_m / 1000.0 + (tower_height_m / 1000.0) ** 2)
    return distance_km


def check_line_of_sight(
    observer_lon: float, observer_lat: float, observer_height_m: float,
    target_lon: float, target_lat: float, target_height_m: float,
    dem_samples: List[Tuple[float, float, float]],  # [(lon, lat, elevation_m), ...]
) -> Tuple[bool, float]:
    """
    检查两点间是否有视线通路。
    返回: (是否可见, 最大遮挡高度差)
    """
    # 计算方位角和距离
    dx = (target_lon - observer_lon) * 111000 * math.cos(math.radians((observer_lat + target_lat) / 2))
    dy = (target_lat - observer_lat) * 111000
    distance = math.sqrt(dx ** 2 + dy ** 2)
    
    if distance < 1:  # 距离 < 1m
        return True, 0.0
    
    # 插值检查路径上每个点的地形遮挡
    steps = max(2, int(distance / 30))  # 每30m采样一次
    max_obstruction = 0.0
    
    for i in range(1, steps):
        t = i / steps
        interp_lon = observer_lon + (target_lon - observer_lon) * t
        interp_lat = observer_lat + (target_lat - observer_lat) * t
        
        # 从DEM获取该点高程（简化：取最近采样点）
        dem_elev = _sample_dem(interp_lon, interp_lat, dem_samples)
        
        # 视线高度（线性插值）
        los_height = observer_height_m + (target_height_m - observer_height_m) * t
        
        obstruction = dem_elev - los_height
        if obstruction > 0:
            max_obstruction = max(max_obstruction, obstruction)
    
    return (max_obstruction == 0, max_obstruction)


def _sample_dem(lon: float, lat: float, dem_samples: List[Tuple[float, float, float]]) -> float:
    """从DEM采样点中获取最近点的高程（简化最近邻插值）"""
    if not dem_samples:
        return 0.0
    
    closest = min(dem_samples, key=lambda s: (s[0] - lon) ** 2 + (s[1] - lat) ** 2)
    return closest[2]
```

        QgsField("name", QVariant.String),

        feat.setAttributes([site.site_id, site.name, site.site_type, site.tower_height, len(site.antennas)])

    for feature in coverage_data.get("features", []):
        coords = feature["geometry"]["coordinates"]
        props = feature["properties"]

        feat = QgsFeature()
        feat.setGeometry(QgsGeometry.fromPointXY(QgsPointXY(coords[0], coords[1])))
        feat.setAttributes([props["rsrp"], props["distanceKm"]])
        dp.addFeature(feat)

    # RSRP分级渲染（通信行业标准）
    rsrp_ranges = [
        ("≥ -75 dBm", -75, 0, QColor(0, 180, 0, 180)),       # 深绿：优秀
        ("-85 ~ -75 dBm", -85, -75, QColor(100, 200, 0, 160)), # 浅绿：良好
        ("-95 ~ -85 dBm", -95, -85, QColor(255, 200, 0, 140)), # 黄色：一般
        ("-105 ~ -95 dBm", -105, -95, QColor(255, 120, 0, 120)),# 橙色：弱覆盖
        ("< -105 dBm", -200, -105, QColor(200, 50, 50, 80)),   # 红色：盲区
    ]

    renderer_ranges = []
    for label, low, high, color in rsrp_ranges:
        symbol = QgsMarkerSymbol.createSimple({
            "name": "square", "color": color.name(), "size": "4",
            "outline_style": "no",
        })
        renderer_range = QgsRendererRange(low, high, symbol, label)
        renderer_ranges.append(renderer_range)

    layer.setRenderer(QgsGraduatedSymbolRenderer("rsrp", renderer_ranges))
    layer.updateExtents()

    return layer

        doc.setDesignVersion(String.valueOf(body.getOrDefault("designVersion", "v1.0")));

        int totalItems = items.size();

        for (Map<String, Object> itemMap : items) {
            BomItem item = new BomItem();
            item.setBomDocumentId(doc.getId());
            item.setItemCode(String.valueOf(itemMap.getOrDefault("itemCode", "")));
            item.setItemName(String.valueOf(itemMap.getOrDefault("itemName", "")));
            item.setCategory(String.valueOf(itemMap.getOrDefault("category", "")));
            item.setSpecification(String.valueOf(itemMap.getOrDefault("specification", "")));
            item.setUnit(String.valueOf(itemMap.getOrDefault("unit", "")));
            BigDecimal qty = new BigDecimal(itemMap.getOrDefault("quantity", 0).toString());
            item.setQuantity(qty);
            item.setRemark(String.valueOf(itemMap.getOrDefault("remark", "")));
            totalQty = totalQty.add(qty);
            bomItemMapper.insert(item);
        }

        doc.setTotalItems(totalItems);
        doc.setTotalQuantity(totalQty);
        bomDocMapper.updateById(doc);

        return Result.success(Map.of("bomId", doc.getId(), "totalItems", totalItems, "totalQuantity", totalQty));
    }

        BomDocument doc = bomDocMapper.selectOne(wrapper);

        List<BomItem> items = bomItemMapper.selectList(
            new LambdaQueryWrapper<BomItem>().eq(BomItem::getBomDocumentId, doc.getId()));
        return Result.success(Map.of("document", doc, "items", items));

"""BOM推送至M04平台API（统一使用requests库）"""
import json
import os
import requests
from pathlib import Path
from bom_generator.bom_model import BomDocument

CONFIG_PATH = Path(__file__).parent.parent / "resources" / "config.json"


def _load_config() -> dict:
    """加载QGIS插件配置"""
    if CONFIG_PATH.exists():
        with open(CONFIG_PATH, "r", encoding="utf-8") as f:
            return json.load(f)
    return {
        "m04_base_url": "http://localhost:8084",
        "jwt_token": "",
        "timeout_seconds": 30,
    }


def push_bom_to_m04(bom: BomDocument) -> dict:
    """将BOM文档推送到M04后端 (API-04)"""
    config = _load_config()
    url = f"{config['m04_base_url']}/api/m04/bom/upload"
    headers = {"Content-Type": "application/json"}
    token = config.get("jwt_token", "")
    if token:
        headers["Authorization"] = f"Bearer {token}"

    resp = requests.post(url, json=bom.to_api_payload(), headers=headers,
                         timeout=config.get("timeout_seconds", 30))
    result = resp.json()
    if result.get("code") == 200:
        return result.get("data", {})
    raise Exception(f"API返回错误: {result.get('message', '未知错误')}")


def push_design_to_m04(geojson_path: str, project_id: str, design_version: str) -> dict:
    """上传设计方案文件到M04后端 (API-01)"""
    config = _load_config()
    url = f"{config['m04_base_url']}/api/m04/design/upload"
    headers = {}
    token = config.get("jwt_token", "")
    if token:
        headers["Authorization"] = f"Bearer {token}"

    with open(geojson_path, "rb") as f:
        resp = requests.post(
            url,
            files={"file": f},
            data={"projectId": project_id, "designVersion": design_version},
            headers=headers,
            timeout=config.get("timeout_seconds", 30),
        )
    result = resp.json()
    if result.get("code") == 200:
        return result.get("data", {})
    raise Exception(f"API返回错误: {result.get('message', '未知错误')}")

public List<Map<String, Object>> executeReview(
        Map<String, Object> designData, Long projectId, Long taskId,
        List<String> dimensions) {
    
    DesignReviewTask task = new DesignReviewTask();
    task.setId(taskId);
    task.setStatus(1);
    taskMapper.updateById(task);

    List<Map<String, Object>> allResults = new ArrayList<>();

    for (ReviewEngineService reviewer : reviewers) {
        // 按需执行：仅运行指定的维度
        if (!dimensions.contains(reviewer.getDimension())) continue;
        
        List<Map<String, Object>> dimResults = reviewer.review(designData, projectId);
        allResults.addAll(dimResults);

        for (Map<String, Object> r : dimResults) {
            ReviewResult entity = new ReviewResult();
            entity.setTaskId(taskId);
            entity.setDimension(String.valueOf(r.getOrDefault("dimension", "")));
            entity.setRuleCode(String.valueOf(r.getOrDefault("ruleCode", "")));
            entity.setRuleName(String.valueOf(r.getOrDefault("ruleName", "")));
            entity.setStatus("PASS".equals(r.get("status")) ? 0 :
                             "WARNING".equals(r.get("status")) ? 1 : 2);
            entity.setSeverity((Integer) r.getOrDefault("severity", 3));
            entity.setLocationGeojson(String.valueOf(r.getOrDefault("locationGeojson", "")));
            entity.setDescription(String.valueOf(r.getOrDefault("description", "")));
            entity.setSuggestion(String.valueOf(r.getOrDefault("suggestion", "")));
            entity.setMeasuredValue(String.valueOf(r.getOrDefault("measuredValue", "")));
            entity.setThresholdValue(String.valueOf(r.getOrDefault("thresholdValue", "")));
            entity.setCreateTime(LocalDateTime.now());
            resultMapper.insert(entity);
        }
    }

    boolean hasFailure = allResults.stream().anyMatch(r -> "FAIL".equals(r.get("status")));
    task.setStatus(hasFailure ? 3 : 2);
    taskMapper.updateById(task);

    return allResults;
}

<script setup>
import { ref, onMounted, watch } from 'vue'
import * as Cesium from 'cesium'

const props = defineProps({ projectId: { type: Number, required: true } })
const cesiumContainer = ref(null)
let viewer = null, dataSource = null

onMounted(() => {
  viewer = new Cesium.Viewer(cesiumContainer.value, {
    terrain: Cesium.Terrain.fromWorldTerrain(),
    animation: false, timeline: false,
  })
  viewer.camera.setView({
    destination: Cesium.Cartesian3.fromDegrees(114.39, 30.50, 8000),

async function loadSites() {
  if (!props.projectId) return
  if (dataSource) viewer.dataSources.remove(dataSource)

  try {
    const res = await fetch(`http://localhost:8084/api/m04/design/${props.projectId}`)
    const geojson = await res.json()
    if (!geojson.features) return

    dataSource = Cesium.GeoJsonDataSource.load(geojson, {
      markerColor: Cesium.Color.fromCssColorString('#1890ff'),
      markerSize: 32,
    })
    viewer.dataSources.add(dataSource)
    viewer.flyTo(dataSource)
  } catch (e) {
    console.error('加载站点数据失败:', e)
  }
}
</script>
